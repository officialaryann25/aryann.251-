Developer Guide

Using the Palette in CSS

Copy any token directly into your own components.

/* ── Import (already in style.css) ──────────────────── */
:root {
  /* PRIMARY — Strawberry Blossom */
  --primary:        #E8486A;   /* buttons, links, badges   */
  --primary-dark:   #C43057;   /* hover / pressed          */
  --primary-light:  #FFE0E9;   /* tint backgrounds         */

  /* SECONDARY — Dark Chocolate */
  --secondary:      #3A1A08;   /* navbar, footer           */
  --secondary-dark: #1E0C02;   /* deepest overlays         */
  --secondary-light:#F2DEC8;   /* warm panels              */

  /* ACCENT — Salted Caramel */
  --accent:         #D48C0F;   /* stars, "Popular" badges  */
  --accent-dark:    #B5750C;   /* caramel hover            */

  /* MINT — Pistachio Mint */
  --mint:           #2AA87A;   /* vegan, fresh, success    */
  --mint-dark:      #1E8A63;   /* mint hover               */
  --mint-light:     #D4F5EA;   /* mint tint                */

  /* BACKGROUNDS */
  --bg-light:       #FFF8EF;   /* page background          */
  --bg-mid:         #FDE8CC;   /* cards, alt sections      */
  --border-color:   #EDD4B0;   /* dividers, input borders  */

  /* TEXT SCALE */
  --text-dark:      #2C1206;   /* headings, body           */
  --text-mid:       #6B3820;   /* secondary text           */
  --text-light:     #A0714D;   /* captions, placeholders   */
}

/* ── Example button ─────────────────────────────────── */
.btn-order-now {
  background: var(--primary);
  color:      #fff;
  padding:    12px 28px;
  border-radius: 9999px;
  font-weight: 700;
  transition: all 0.3s ease;
}
.btn-order-now:hover {
  background: var(--primary-dark);
  box-shadow: 0 6px 20px rgba(232,72,106,0.40);
  transform:  translateY(-2px);
}

/* ── Example card ───────────────────────────────────── */
.ice-cream-card {
  background:    var(--bg-white);
  border:        1px solid var(--border-color);
  border-top:    4px solid var(--primary);
  border-radius: 20px;
  padding:       24px;
  box-shadow:    0 4px 20px rgba(232,72,106,0.12);
}
.ice-cream-card:hover {
  transform:  translateY(-6px);
  box-shadow: 0 8px 40px rgba(232,72,106,0.18);
}
